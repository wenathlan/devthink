   113→   113→function stableId(): string {
   114→   114→  const alphabet = "0123456789abcdefghjkmnpqrstvwxyz";
   115→   115→  const bytes = new Uint8Array(10);
   116→   116→  globalThis.crypto?.getRandomValues?.(bytes);
   117→   117→  const token = Array.from(bytes, (value) => alphabet[value & 31]).join("") || Math.random().toString(36).slice(2, 12);
   118→   118→  return `l_${token}`;
   119→   119→}

   130→   130→function formatTime(value: string): string {
   131→   131→  const timestamp = new Date(value).getTime();
   132→   132→  if (!Number.isFinite(timestamp)) return "local";
   133→   133→  const seconds = Math.max(0, Math.floor((Date.now() - timestamp) / 1000));
   134→   134→  return seconds < 60 ? "now" : `${Math.floor(seconds / 60)}m`;
   135→   135→}

   602→   602→  async function sendMessage(event: FormEvent) {
   603→   603→    event.preventDefault();
   604→   604→    const prompt = draft.trim();
   605→   605→    if (!prompt) return;
   606→   606→    const userId = stableId();
   607→   607→    const assistantId = stableId();
   608→   608→    setMessages((current) => [
   609→   609→      ...current,
   610→   610→      {
   611→   611→        id: userId,
   612→   612→        workspaceId: route.workspaceId,
   613→   613→        sessionId: route.sessionId,
   614→   614→        tabId: route.tabId,
   615→   615→        sectionId: route.sectionId,
   616→   616→        role: "user",
   617→   617→        title: "Request staged",
   618→   618→        body: prompt,
   619→   619→        time: "now",
   620→   620→      },
   621→   621→      {
   622→   622→        id: assistantId,
   623→   623→        workspaceId: route.workspaceId,
   624→   624→        sessionId: route.sessionId,
   625→   625→        tabId: route.tabId,
   626→   626→        sectionId: route.sectionId,
   627→   627→        role: "assistant",
   628→   628→        title: "Gateway response",
   629→   629→        body: gatewayUrl
   630→   630→          ? "Connecting to the local DevThink gateway…"
   631→   631→          : "Add a user-configured gateway URL to dispatch this request through the local DevThink CLI.",
   632→   632→        time: "pending",
   633→   633→      },
   634→   634→    ]);
   635→   635→    setDraft("");
   636→   636→    if (!gatewayUrl || !browserToken) {
   637→   637→      void saveBrowserMessages(route, [
   638→   638→        { id: userId, role: "user", content: prompt },
   639→   639→        {
   640→   640→          id: assistantId,
   641→   641→          role: "assistant",
   642→   642→          content: "Add a user-configured gateway URL to dispatch this request through the local DevThink CLI.",
   643→   643→        },
   644→   644→      ]);
   645→   645→      return toast(
   646→   646→        "Request is staged in this browser. Pair the DevThink gateway to persist and stream it through the CLI.",
   647→   647→      );
   648→   648→    }
   649→   649→    try {
   650→   650→      const response = await fetch(`${gatewayUrl}/chat`, {
   651→   651→        method: "POST",
   652→   652→        headers: { "content-type": "application/json", ...browserHeaders },
   653→   653→        body: JSON.stringify({
   654→   654→          workspaceId: route.workspaceId,
   655→   655→          sessionId: route.sessionId,
   656→   656→          tabId: route.tabId,
   657→   657→          sectionId: route.sectionId,
   658→   658→          provider: selectedProvider,
   659→   659→          model: provider.model,
   660→   660→          messages: [{ role: "user", content: prompt }],
   661→   661→        }),
   662→   662→      });
   663→   663→      if (!response.ok || !response.body) throw new Error("Gateway request failed.");
   664→   664→      const reader = response.body.getReader();
   665→   665→      const decoder = new TextDecoder();
   666→   666→      let pending = "";
   667→   667→      while (true) {
   668→   668→        const next = await reader.read();
   669→   669→        if (next.done) break;
   670→   670→        pending += decoder.decode(next.value, { stream: true });
   671→   671→        const boundary = pending.lastIndexOf("\n\n");
   672→   672→        if (boundary < 0) continue;
   673→   673→        const frames = sseEvents(pending.slice(0, boundary));
   674→   674→        pending = pending.slice(boundary + 2);
   675→   675→        for (const frame of frames) {
   676→   676→          if (frame.type === "identity")
   677→   677→            navigate({
   678→   678→              workspaceId: String(frame.data.workspaceId),
   679→   679→              sessionId: String(frame.data.sessionId),
   680→   680→              tabId: String(frame.data.tabId),
   681→   681→              sectionId: String(frame.data.sectionId),
   682→   682→            });
   683→   683→          if (frame.type === "text")
   684→   684→            setMessages((current) =>
   685→   685→              current.map((message) =>
   686→   686→                message.id === assistantId
   687→   687→                  ? {
   688→   688→                      ...message,
   689→   689→                      body: `${message.body === "Connecting to the local DevThink gateway…" ? "" : message.body}${String(frame.data.text || "")}`,
   690→   690→                      time: "live",
   691→   691→                    }
   692→   692→                  : message,
   693→   693→              ),
   694→   694→            );
   695→   695→          if (frame.type === "persisted")
   696→   696→            setMessages((current) =>
   697→   697→              current.map((message) =>
   698→   698→                message.id === assistantId
   699→   699→                  ? { ...message, id: String(frame.data.messageId || assistantId), time: "saved" }
   700→   700→                  : message,
   701→   701→              ),
   702→   702→            );
   703→   703→        }
   704→   704→      }
   705→   705→      const saved = await fetch(`${gatewayUrl}/sessions/${encodeURIComponent(route.sessionId)}`, {
   706→   706→        headers: browserHeaders,
   707→   707→      }).then((next) =>
   708→   708→        next.ok ? (next.json() as Promise<GatewaySession>) : Promise.reject(new Error("Session cache unavailable.")),
   709→   709→      );
   710→   710→      await cacheGatewaySession(saved);
   711→   711→      toast("Response persisted by the local DevThink gateway.");
   712→   712→    } catch {
   713→   713→      const failure =
   714→   714→        "The local gateway could not complete this request. Verify the gateway URL, provider configuration, and the explicit provider credential.";
   715→   715→      setMessages((current) =>
   716→   716→        current.map((message) => (message.id === assistantId ? { ...message, body: failure, time: "error" } : message)),
   717→   717→      );
   718→   718→      void saveBrowserMessages(route, [
   719→   719→        { id: userId, role: "user", content: prompt },
   720→   720→        { id: assistantId, role: "assistant", content: failure },
   721→   721→      ]);
   722→   722→      toast("Gateway request was not completed.");
   723→   723→    }
   724→   724→  }
