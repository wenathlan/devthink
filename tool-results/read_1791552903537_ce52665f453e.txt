    60→/* ============================ theme context ============================ */
    61→
    62→type Theme = 'light' | 'dark';
    63→
    64→interface ThemeContextType {
    65→        theme: Theme;
    66→        toggleTheme?: () => void;
    67→        switchable: boolean;
    68→}
    69→
    70→const ThemeContext = createContext<ThemeContextType | undefined>(undefined);
    71→
    72→interface ThemeProviderProps {
    73→        children: React.ReactNode;
    74→        defaultTheme?: Theme;
    75→        switchable?: boolean;
    76→}
    77→
    78→export function ThemeProvider({
    79→        children,
    80→        defaultTheme = 'light',
    81→        switchable = false,
    82→}: ThemeProviderProps) {
    83→        const [theme, setTheme] = useState<Theme>(() => {
    84→                if (switchable) {
    85→                        const stored = localStorage.getItem('theme');
    86→                        return (stored as Theme) || defaultTheme;
    87→                }
    88→                return defaultTheme;
    89→        });
    90→
    91→        useEffect(() => {
    92→                const root = document.documentElement;
    93→                if (theme === 'dark') {
    94→                        root.classList.add('dark');
    95→                } else {
    96→                        root.classList.remove('dark');
    97→                }
    98→
    99→                if (switchable) {
   100→                        localStorage.setItem('theme', theme);
   101→                }
   102→        }, [theme, switchable]);
   103→
   104→        const toggleTheme = switchable
   105→                ? () => {
   106→                                setTheme((prev) => (prev === 'light' ? 'dark' : 'light'));
   107→                        }
   108→                : undefined;
   109→
   110→        return (
   111→                <ThemeContext.Provider value={{ theme, toggleTheme, switchable }}>
   112→                        {children}
   113→                </ThemeContext.Provider>
   114→        );
   115→}
   116→
   117→export function useTheme() {
   118→        const context = useContext(ThemeContext);
   119→        if (!context) {
   120→                throw new Error('useTheme must be used within ThemeProvider');
   121→        }
   122→        return context;
   123→}

   531→/**
   532→ * the theme flip of the rail foot: flips the document theme class, stores
   533→ * nothing (the in-memory doctrine of the window chrome).
   534→ *
   535→ * @returns the toggle element.
   536→ */
   537→function ThemeToggle() {
   538→        const [theme, setTheme] = useState<Theme>(() =>
   539→                typeof document === 'undefined' || document.documentElement.classList.contains('dark')
   540→                        ? 'dark'
   541→                        : 'light',
   542→        );
   543→        /** flips the dark class on the root and mirrors it into the label. */
   544→        function toggle() {
   545→                const next = theme === 'dark' ? 'light' : 'dark';
   546→                document.documentElement.classList.toggle('dark', next === 'dark');
   547→                setTheme(next);
   548→        }
   549→        return (
   550→                <button type="button" className="winapp__theme" aria-label="toggle theme" onClick={toggle}>
   551→                        {theme === 'dark' ? (
   552→                                <Sun size={14} aria-hidden="true" />
   553→                        ) : (
   554→                                <Moon size={14} aria-hidden="true" />
   555→                        )}
   556→                        {theme}
   557→                </button>
   558→        );
   559→}

   576→ * the application window: title bar (brand, drag, caption buttons), the
   577→ * SECTIONED page rail and the routed stage, floating over the Mica
   578→ * backdrop.
   579→ *
   580→ * @param children the routed page.
   581→ * @returns the window element.
   582→ */
   583→export function Shell({ children, themeButton = true }: ShellProps) {
   584→        const [location, navigate] = useLocation();
   585→        const [maximized, setMaximized] = useState(false);
   586→        const [minimized, setMinimized] = useState(false);
   587→        const [drag, setDrag] = useState({ x: 0, y: 0 });
   588→        const trackRef = useRef<DragTrack | null>(null);
   589→
   590→        /** maximizes over the backdrop or restores the windowed frame (the drag offset resets). */
   591→        function toggleMaximize() {
   592→                setDrag({ x: 0, y: 0 });
   593→                setMaximized((value) => !value);
   594→        }
   595→
   596→        /** arms a drag when the pointer presses the empty bar (never the caption buttons). */
   597→        function onTitlePointerDown(event: ReactPointerEvent<HTMLElement>) {
   598→                if (maximized) return;
   599→                if ((event.target as HTMLElement).closest('button, a, input')) return;
   600→                trackRef.current = {
   601→                        pointerId: event.pointerId,
   602→                        originX: event.clientX,
   603→                        originY: event.clientY,
   604→                        baseX: drag.x,
   605→                        baseY: drag.y,
   606→                };
   607→                event.currentTarget.setPointerCapture(event.pointerId);
   608→        }
   609→
   610→        /** rides the pointer while a drag session is armed. */
   611→        function onTitlePointerMove(event: ReactPointerEvent<HTMLElement>) {
   612→                const track = trackRef.current;
   613→                if (!track || track.pointerId !== event.pointerId) return;
   614→                setDrag(
   615→                        clampDrag(
   616→                                track.baseX + event.clientX - track.originX,
   617→                                track.baseY + event.clientY - track.originY,
   618→                        ),
   619→                );
   620→        }
   621→
   622→        /** releases the drag session. */
   623→        function onTitlePointerUp(event: ReactPointerEvent<HTMLElement>) {
   624→                const track = trackRef.current;
   625→                if (!track || track.pointerId !== event.pointerId) return;
   626→                trackRef.current = null;
   627→                event.currentTarget.releasePointerCapture(event.pointerId);
   628→        }
   629→
   630→        /** the Windows affordance: a double click on the empty bar toggles maximize. */
   631→        function onTitleDoubleClick(event: ReactMouseEvent<HTMLElement>) {
   632→                if ((event.target as HTMLElement).closest('button, a, input')) return;
   633→                toggleMaximize();
   634→        }
   635→
   636→        // the active rail entry: the bare path owns home, every other page owns its route
   637→        const isActive = (href: string): boolean =>
   638→                href === '/' ? location === '/' : location === href || location.startsWith(`${href}/`);
   639→
   640→        return (
   641→                <div className="appframe" data-maximized={maximized ? 'true' : undefined}>
   642→                        <section
   643→                                className="winapp"
   644→                                hidden={minimized}
   645→                                data-maximized={maximized ? 'true' : undefined}
   646→                                aria-label="saddle"
   647→                                style={{ '--winapp-x': `${drag.x}px`, '--winapp-y': `${drag.y}px` } as CSSProperties}
   648→                        >
   649→                                <header
   650→                                        className="winapp__title"
   651→                                        data-dragging={trackRef.current ? 'true' : undefined}
   652→                                        onPointerDown={onTitlePointerDown}
   653→                                        onPointerMove={onTitlePointerMove}
   654→                                        onPointerUp={onTitlePointerUp}
   655→                                        onPointerCancel={onTitlePointerUp}
   656→                                        onDoubleClick={onTitleDoubleClick}
   657→                                >
   658→                                        <div className="winapp__brand">
   659→                                                <SaddleMark size={20} hidden />
   660→                                                <span className="winapp__name">saddle</span>
   661→                                                <span className="winapp__role">{APP_ROLE}</span>
   662→                                        </div>
   663→                                        <div className="winapp__caption">
   664→                                                <button
   665→                                                        type="button"
   666→                                                        className="winapp__cap"
   667→                                                        aria-label="Minimize"
   668→                                                        onClick={() => setMinimized(true)}
   669→                                                >
   670→                                                        <Minus size={14} aria-hidden="true" />
   671→                                                </button>
   672→                                                <button
   673→                                                        type="button"
   674→                                                        className="winapp__cap"
   675→                                                        aria-label={maximized ? 'Restore' : 'Maximize'}
   676→                                                        onClick={toggleMaximize}
   677→                                                >
   678→                                                        {maximized ? (
   679→                                                                <Copy size={12} aria-hidden="true" />
   680→                                                        ) : (
   681→                                                                <Square size={11} aria-hidden="true" />
   682→                                                        )}
   683→                                                </button>
   684→                                                <button
   685→                                                        type="button"
   686→                                                        className="winapp__cap winapp__cap--close"
   687→                                                        aria-label="Close"
   688→                                                        onClick={() => navigate(RESTART_ROUTE)}
   689→                                                >
   690→                                                        <X size={14} aria-hidden="true" />
   691→                                                </button>
   692→                                        </div>
   693→                                </header>
   694→
   695→                                <div className="winapp__body">
   696→                                        <nav className="winapp__rail" aria-label="Pages">
   697→                                                {NAV_SECTIONS.map((section) => (
   698→                                                        <div key={section.id} className="winapp__section">
   699→                                                                <p className="winapp__sectionlabel" aria-hidden="true">
   700→                                                                        {section.label}
   701→                                                                </p>
   702→                                                                {section.apps.map((app) => {
   703→                                                                        const Icon = app.icon;
   704→                                                                        return (
   705→                                                                                <Link
   706→                                                                                        key={app.href}
   707→                                                                                        href={app.href}
   708→                                                                                        className="winapp__raillink"
   709→                                                                                        title={app.detail}
   710→                                                                                        aria-label={app.label}
   711→                                                                                        aria-current={isActive(app.href) ? 'page' : undefined}
   712→                                                                                >
   713→                                                                                        <Icon size={16} strokeWidth={1.7} aria-hidden="true" />
   714→                                                                                        <span className="winapp__raillabel">{app.label}</span>
   715→                                                                                </Link>
   716→                                                                        );
   717→                                                                })}
   718→                                                        </div>
   719→                                                ))}
   720→                                                {themeButton ? (
   721→                                                        <div className="winapp__railfoot">
   722→                                                                <ThemeToggle />
   723→                                                        </div>
   724→                                                ) : null}
   725→                                        </nav>
   726→                                        <div className="winapp__stage">
   727→                                                <main className="shell">{children}</main>
   728→                                        </div>
   729→                                </div>
   730→                        </section>
   731→
   732→                        {minimized && (
   733→                                <button
   734→                                        type="button"
   735→                                        className="winchip"
   736→                                        aria-label="Restore saddle"
   737→                                        onClick={() => setMinimized(false)}
   738→                                >
   739→                                        <SaddleMark size={16} hidden />
   740→                                        <span>saddle</span>
   741→                                </button>
   742→                        )}
   743→                </div>
   744→        );
   745→}
