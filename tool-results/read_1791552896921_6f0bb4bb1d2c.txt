     1→/**
     2→ * appicons.tsx — the premium drawn icon set of the Sol shell. One hand-drawn
     3→ * SVG component per desktop app (Sol/shell/appregistry.ts), exported through
     4→ * the APP_ICONS map the app tile reads by the registry `iconset` key.
     5→ *
     6→ * Every icon is built in depth layers over its own guide color (the "story"):
     7→ * a gradient squircle with a soft top gloss, a mid layer of blurred story
     8→ * orbs over a pedestal band, and the glyph itself in thick ivory strokes with
     9→ * a translucent filled backing and a blurred drop shadow. The finishing
    10→ * follows the house reference (the 3D stage icon): two blurred inner
    11→ * contours, a blurred contact ellipse at the base, a discrete film grain, a
    12→ * two-layer drop shadow in the story color, a sheen band that sweeps once on
    13→ * hover, a contact glow that rises and breathes, and a gentle perspective
    14→ * tilt — all plain CSS transitions/animations on transform/opacity/filter,
    15→ * with @property registered custom properties (--dt-fan/--dt-glow) easing at
    16→ * the house cubic-bezier and -webkit- prefixes on every 3D/filter path.
    17→ * No decorative dots and no microcopy: the interaction exists, the icon
    18→ * speaks for itself. prefers-reduced-motion and coarse pointers switch the
    19→ * loop and the dramatic hover off.
    20→ *
    21→ * The optical scale every glyph answers to (the audit ladder): one stroke
    22→ * hierarchy — STROKE_BRIGHT 5.5 for the primary outline, STROKE 5 for the
    23→ * base, STROKE_SOFT 4.5 for secondary details, STROKE_HAIR 4 for fine inner
    24→ * contours; tiny windows at radius 2, panes and crate at 2.5, one bar radius
    25→ * (3), container radii 5–6.5 and the board at 8; and one inner-padding band,
    26→ * glyphs living within ~26–70 of the 96-unit face. The identity of every
    27→ * glyph stays untouched; only outliers are normalized onto the ladder.
    28→ */
    29→import { type ComponentType, type CSSProperties, type ReactElement, useEffect } from "react";

   156→let iconCssReady = false;
   157→
   158→/** Injects the shared stylesheet of the set exactly once per document. */
   159→function ensureIconCss(): void {
   160→  if (iconCssReady || typeof document === "undefined") return;
   161→  iconCssReady = true;
   162→  const tag = document.createElement("style");
   163→  tag.setAttribute("data-dt-icons", "");
   164→  tag.textContent = ICON_CSS;
   165→  document.head.appendChild(tag);
   166→}
   167→
   168→let iconMotionChecked = false;
   169→
   170→/**
   171→ * Registers the animatable custom properties and gates the custom-property
   172→ * loops behind the dt-icon-fx class (browsers without the @property API
   173→ * never run them, so nothing flips discretely).
   174→ */
   175→function useIconMotion(): void {
   176→  useEffect(() => {
   177→    if (iconMotionChecked) return;
   178→    iconMotionChecked = true;
   179→    const css = typeof CSS === "undefined" ? null : (CSS as unknown as { registerProperty?: (p: HoudiniProp) => void });
   180→    if (css && typeof css.registerProperty === "function") {
   181→      for (const prop of HOUDINI_PROPS) {
   182→        try {
   183→          css.registerProperty(prop);
   184→        } catch {
   185→          /* already registered by the stylesheet */
   186→        }
   187→      }
   188→      document.documentElement.classList.add("dt-icon-fx");
   189→    }
   190→  }, []);
   191→}

   193→/* ------------------------------- the factory ------------------------------- */
   194→type AppIconSpec = {
   195→  /** the guide color of the app (the story the icon catalog row declares) */
   196→  story: string;
   197→  /** the deep shade the gradient settles into */
   198→  deep: string;
   199→  /** the warm subtone of the mid layer, contours and contact ellipse */
   200→  soft: string;
   201→  /** the drawn glyph: thick ivory strokes with a translucent filled backing */
   202→  glyph: ReactElement;
   203→};
   204→
   205→/** The asymmetric squircle of the tile face: tighter shoulders, heavier base. */
   206→const SQUIRCLE = "M22 0 L74 0 Q96 0 96 22 L96 66 Q96 96 66 96 L30 96 Q0 96 0 66 L0 22 Q0 0 22 0 Z";
   207→
   208→/**
   209→ * Builds one premium icon component over a story palette and glyph.
   210→ *
   211→ * @param key the registry id the iconset field points at (namespaces every id)
   212→ * @param spec the story colors and the drawn glyph of the app
   213→ */
   214→function defineAppIcon(key: string, spec: AppIconSpec): ComponentType<AppIconProps> {
   215→  const uid = `dti-${key}`;
   216→  function AppIcon({ size }: AppIconProps) {
   217→    ensureIconCss();
   218→    useIconMotion();
   219→    const vars = {
   220→      "--story-glow": `${spec.story}73`,
   221→      "--story-shadow": `${spec.deep}42`,
   222→      "--story-shadow-soft": `${spec.deep}24`,
   223→    } as CSSProperties;
   224→    if (size !== undefined) {
   225→      vars.width = size;
   226→      vars.height = size;
   227→    }
   228→    return (
   229→      <span className="dtIcon" style={vars} aria-hidden="true">
   230→        <span className="dtIconGlow" />
   231→        <svg className="dtIconSvg" viewBox="0 0 96 96" aria-hidden="true" focusable="false">
   232→          <defs>
   233→            <linearGradient id={`${uid}-bg`} x1="0" y1="0" x2="0" y2="1">
   234→              <stop offset="0" stopColor={spec.story} />
   235→              <stop offset=".6" stopColor={spec.story} />
   236→              <stop offset="1" stopColor={spec.deep} />
   237→            </linearGradient>
   238→            <linearGradient id={`${uid}-gloss`} x1="0" y1="0" x2="0" y2="1">
   239→              <stop offset="0" stopColor="#ffffff" stopOpacity=".32" />
   240→              <stop offset="1" stopColor="#ffffff" stopOpacity="0" />
   241→            </linearGradient>
   242→            <radialGradient id={`${uid}-orb`} cx=".5" cy=".5" r=".5">
   243→              <stop offset="0" stopColor={spec.soft} stopOpacity=".9" />
   244→              <stop offset=".35" stopColor={spec.soft} stopOpacity=".5" />
   245→              <stop offset="1" stopColor={spec.soft} stopOpacity="0" />
   246→            </radialGradient>
   247→            <linearGradient id={`${uid}-edge-l`} x1="0" y1="0" x2="0" y2="1">
   248→              <stop offset="0" stopColor={spec.soft} stopOpacity="0" />
   249→              <stop offset=".45" stopColor={spec.soft} stopOpacity=".3" />
   250→              <stop offset="1" stopColor={spec.soft} stopOpacity=".7" />
   251→            </linearGradient>
   252→            <linearGradient id={`${uid}-edge-d`} x1="0" y1="0" x2="0" y2="1">
   253→              <stop offset="0" stopColor={spec.soft} stopOpacity="0" />
   254→              <stop offset=".5" stopColor={spec.soft} stopOpacity=".22" />
   255→              <stop offset="1" stopColor={spec.soft} stopOpacity=".55" />
   256→            </linearGradient>
   257→            <linearGradient id={`${uid}-sheen`} x1="0" y1="0" x2="1" y2="1">
   258→              <stop offset="0" stopColor="#ffffff" stopOpacity="0" />
   259→              <stop offset=".45" stopColor="#ffffff" stopOpacity=".5" />
   260→              <stop offset=".55" stopColor="#ffffff" stopOpacity=".5" />
   261→              <stop offset="1" stopColor="#ffffff" stopOpacity="0" />
   262→            </linearGradient>
   263→            <filter id={`${uid}-soft`} x="-40%" y="-40%" width="180%" height="180%">
   264→              <feGaussianBlur stdDeviation="3" />
   265→            </filter>
   266→            <filter id={`${uid}-wide`} x="-60%" y="-60%" width="220%" height="220%">
   267→              <feGaussianBlur stdDeviation="8" />
   268→            </filter>
   269→            <filter id={`${uid}-lift`} x="-40%" y="-40%" width="180%" height="180%">
   270→              <feDropShadow dx="0" dy="2" stdDeviation="2" floodColor={spec.deep} floodOpacity=".38" />
   271→            </filter>
   272→            <filter id={`${uid}-grain`} x="0%" y="0%" width="100%" height="100%">
   273→              <feTurbulence type="fractalNoise" baseFrequency="0.9" numOctaves="2" stitchTiles="stitch" result="n" />
   274→              <feColorMatrix in="n" type="saturate" values="0" />
   275→              <feComposite operator="in" in2="SourceGraphic" />
   276→            </filter>
   277→            <clipPath id={`${uid}-clip`}>
   278→              <path d={SQUIRCLE} />
   279→            </clipPath>
   280→          </defs>
   281→
   282→          {/* the face: gradient squircle with a soft top gloss */}
   283→          <path d={SQUIRCLE} fill={`url(#${uid}-bg)`} />
   284→          <path d={SQUIRCLE} fill={`url(#${uid}-gloss)`} opacity=".5" />
   285→
   286→          <g clipPath={`url(#${uid}-clip)`}>
   287→            {/* the mid layer: story orbs breathing over a pedestal band */}
   288→            <g className="dtIconMid">
   289→              <circle cx="48" cy="40" r="25" fill={`url(#${uid}-orb)`} filter={`url(#${uid}-wide)`} opacity=".85" />
   290→              <rect x="-12" y="56" width="120" height="44" fill={spec.soft} opacity=".3" filter={`url(#${uid}-soft)`} />
   291→            </g>
   292→
   293→            {/* two blurred inner contours of the squircle */}
   294→            <path
   295→              d={SQUIRCLE}
   296→              fill="none"
   297→              stroke={`url(#${uid}-edge-d)`}
   298→              strokeWidth="6"
   299→              filter={`url(#${uid}-wide)`}
   300→              opacity=".55"
   301→              transform="translate(1.4 1.9) scale(0.97)"
   302→            />
   303→            <path
   304→              d={SQUIRCLE}
   305→              fill="none"
   306→              stroke={`url(#${uid}-edge-l)`}
   307→              strokeWidth="2.5"
   308→              filter={`url(#${uid}-soft)`}
   309→              opacity=".5"
   310→            />
   311→
   312→            {/* the contact ellipse at the base */}
   313→            <ellipse
   314→              cx="48"
   315→              cy="94"
   316→              rx="30"
   317→              ry="7"
   318→              fill={`url(#${uid}-orb)`}
   319→              filter={`url(#${uid}-soft)`}
   320→              opacity=".55"
   321→            />
   322→
   323→            {/* the sheen band sweeping once on hover */}
   324→            <g className="dtIconSheenBand">
   325→              <rect x="-11" y="-24" width="26" height="144" fill={`url(#${uid}-sheen)`} transform="skewX(-16)" />
   326→            </g>
   327→
   328→            {/* the discrete film grain */}
   329→            <path d={SQUIRCLE} fill="#ffffff" filter={`url(#${uid}-grain)`} opacity=".08" />
   330→          </g>
   331→
   332→          {/* the glyph: thick ivory strokes lifting toward the viewer */}
   333→          <g filter={`url(#${uid}-lift)`}>
   334→            <g
   335→              className="dtIconGlyph"
   336→              fill="none"
   337→              stroke={IVORY}
   338→              strokeWidth={STROKE}
   339→              strokeLinecap="round"
   340→              strokeLinejoin="round"
   341→            >
   342→              {spec.glyph}
   343→            </g>
   344→          </g>
   345→        </svg>
   346→      </span>
   347→    );
   348→  }
   349→  AppIcon.displayName = `${key}AppIcon`;
   350→  return AppIcon;
   351→}
