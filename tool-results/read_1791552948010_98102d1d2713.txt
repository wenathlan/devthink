   113→function stableId(): string {
   114→  const alphabet = "0123456789abcdefghjkmnpqrstvwxyz";
   115→  const bytes = new Uint8Array(10);
   116→  globalThis.crypto?.getRandomValues?.(bytes);
   117→  const token = Array.from(bytes, (value) => alphabet[value & 31]).join("") || Math.random().toString(36).slice(2, 12);
   118→  return `l_${token}`;
   119→}

   121→function gatewayFromSearch(search: string): string | undefined {
   122→  const configured = new URLSearchParams(search).get("gateway") || import.meta.env.VITE_DEVTHINK_GATEWAY_URL;

   130→function formatTime(value: string): string {
   131→  const timestamp = new Date(value).getTime();
   132→  if (!Number.isFinite(timestamp)) return "local";
   133→  const seconds = Math.max(0, Math.floor((Date.now() - timestamp) / 1000));
   134→  return seconds < 60 ? "now" : `${Math.floor(seconds / 60)}m`;
   135→}

   538→          sectionId: tab.sectionId || "chat",
   539→        },
   540→        { label: tab.label, provider: tab.provider, sectionId: tab.sectionId },
   541→      );
   542→      navigate({ tabId: tab.id, sectionId: tab.sectionId || "chat" });
   543→      toast("A clean local session has been opened.");
   544→    };
   545→    if (!gatewayUrl)
   546→      return apply({
   547→        id: localId,
   548→        workspaceId: route.workspaceId,
   549→        sessionId: route.sessionId,
   550→        label: "new route",
   551→        provider: selectedProvider,
   552→        sectionId: "chat",
   553→      });
   554→    if (!browserToken) return toast("Pair the local CLI before creating a persisted tab.");
   555→    void fetch(`${gatewayUrl}/sessions/${encodeURIComponent(route.sessionId)}/tabs`, {
   556→      method: "POST",
   557→      headers: { "content-type": "application/json", ...browserHeaders },
   558→      body: JSON.stringify({ label: "new route", provider: selectedProvider, sectionId: "chat" }),
   559→    })
   560→      .then(async (response) =>
   561→        response.ok ? (response.json() as Promise<GatewaySession>) : Promise.reject(new Error("Tab creation failed.")),
   562→      )
   563→      .then((session) => {
   564→        const tab = session.tabs.find((item) => item.id === session.activeTabId);
   565→        if (!tab) throw new Error("Created tab unavailable.");
   566→        apply({
   567→          id: tab.id,
   568→          workspaceId: tab.workspaceId,
   569→          sessionId: tab.sessionId,
   570→          label: tab.label,
   571→          provider: tab.provider || selectedProvider,
   572→          sectionId: tab.sectionId,
   573→        });
   574→      })
   575→      .catch(() =>
   576→        apply({
   577→          id: localId,
   578→          workspaceId: route.workspaceId,
   579→          sessionId: route.sessionId,
   580→          label: "new route",
   581→          provider: selectedProvider,
   582→          sectionId: "chat",
   583→        }),
   584→      );

   602→  async function sendMessage(event: FormEvent) {
   603→    event.preventDefault();
   604→    const prompt = draft.trim();
   605→    if (!prompt) return;
   606→    const userId = stableId();
   607→    const assistantId = stableId();
   608→    setMessages((current) => [
   609→      ...current,
   610→      {
   611→        id: userId,
   612→        workspaceId: route.workspaceId,
   613→        sessionId: route.sessionId,
   614→        tabId: route.tabId,
   615→        sectionId: route.sectionId,
   616→        role: "user",
   617→        title: "Request staged",
   618→        body: prompt,
   619→        time: "now",
   620→      },
   621→      {
   622→        id: assistantId,
   623→        workspaceId: route.workspaceId,
   624→        sessionId: route.sessionId,
   625→        tabId: route.tabId,
   626→        sectionId: route.sectionId,
   627→        role: "assistant",
   628→        title: "Gateway response",
   629→        body: gatewayUrl
   630→          ? "Connecting to the local DevThink gateway…"
   631→          : "Add a user-configured gateway URL to dispatch this request through the local DevThink CLI.",
   632→        time: "pending",
   633→      },
   634→    ]);
   635→    setDraft("");
   636→    if (!gatewayUrl || !browserToken) {
   637→      void saveBrowserMessages(route, [
   638→        { id: userId, role: "user", content: prompt },
   639→        {
   640→          id: assistantId,
   641→          role: "assistant",
   642→          content: "Add a user-configured gateway URL to dispatch this request through the local DevThink CLI.",
   643→        },
   644→      ]);
   645→      return toast(
   646→        "Request is staged in this browser. Pair the DevThink gateway to persist and stream it through the CLI.",
   647→      );
   648→    }
   649→    try {
   650→      const response = await fetch(`${gatewayUrl}/chat`, {
   651→        method: "POST",
   652→        headers: { "content-type": "application/json", ...browserHeaders },
   653→        body: JSON.stringify({
   654→          workspaceId: route.workspaceId,
   655→          sessionId: route.sessionId,
   656→          tabId: route.tabId,
   657→          sectionId: route.sectionId,
   658→          provider: selectedProvider,
   659→          model: provider.model,
   660→          messages: [{ role: "user", content: prompt }],
   661→        }),
   662→      });
   663→      if (!response.ok || !response.body) throw new Error("Gateway request failed.");
   664→      const reader = response.body.getReader();
   665→      const decoder = new TextDecoder();
   666→      let pending = "";
   667→      while (true) {
   668→        const next = await reader.read();
   669→        if (next.done) break;
   670→        pending += decoder.decode(next.value, { stream: true });
   671→        const boundary = pending.lastIndexOf("\n\n");
   672→        if (boundary < 0) continue;
   673→        const frames = sseEvents(pending.slice(0, boundary));
   674→        pending = pending.slice(boundary + 2);
   675→        for (const frame of frames) {
   676→          if (frame.type === "identity")
   677→            navigate({
   678→              workspaceId: String(frame.data.workspaceId),
   679→              sessionId: String(frame.data.sessionId),
   680→              tabId: String(frame.data.tabId),
   681→              sectionId: String(frame.data.sectionId),
   682→            });
   683→          if (frame.type === "text")
   684→            setMessages((current) =>
   685→              current.map((message) =>
   686→                message.id === assistantId
   687→                  ? {
   688→                      ...message,
   689→                      body: `${message.body === "Connecting to the local DevThink gateway…" ? "" : message.body}${String(frame.data.text || "")}`,
   690→                      time: "live",
   691→                    }
   692→                  : message,
   693→              ),
   694→            );
   695→          if (frame.type === "persisted")
   696→            setMessages((current) =>
   697→              current.map((message) =>
   698→                message.id === assistantId
   699→                  ? { ...message, id: String(frame.data.messageId || assistantId), time: "saved" }
   700→                  : message,
   701→              ),
   702→            );
   703→        }
   704→      }
   705→      const saved = await fetch(`${gatewayUrl}/sessions/${encodeURIComponent(route.sessionId)}`, {
   706→        headers: browserHeaders,
   707→      }).then((next) =>
   708→        next.ok ? (next.json() as Promise<GatewaySession>) : Promise.reject(new Error("Session cache unavailable.")),
   709→      );
   710→      await cacheGatewaySession(saved);
   711→      toast("Response persisted by the local DevThink gateway.");
   712→    } catch {
   713→      const failure =
   714→        "The local gateway could not complete this request. Verify the gateway URL, provider configuration, and the explicit provider credential.";
   715→      setMessages((current) =>
   716→        current.map((message) => (message.id === assistantId ? { ...message, body: failure, time: "error" } : message)),
   717→      );
   718→      void saveBrowserMessages(route, [
   719→        { id: userId, role: "user", content: prompt },
   720→        { id: assistantId, role: "assistant", content: failure },
   721→      ]);
   722→      toast("Gateway request was not completed.");
   723→    }
   724→  }
