    44→/* ============================ theme context ============================ */
    45→
    46→type Theme = "light" | "dark";
    47→
    48→interface ThemeContextType {
    49→  theme: Theme;
    50→  toggleTheme?: () => void;
    51→  switchable: boolean;
    52→}
    53→
    54→const ThemeContext = createContext<ThemeContextType | undefined>(undefined);
    55→
    56→interface ThemeProviderProps {
    57→  children: React.ReactNode;
    58→  defaultTheme?: Theme;
    59→  switchable?: boolean;
    60→}
    61→
    62→export function ThemeProvider({
    63→  children,
    64→  defaultTheme = "light",
    65→  switchable = false,
    66→}: ThemeProviderProps) {
    67→  const [theme, setTheme] = useState<Theme>(() => {
    68→    if (switchable) {
    69→      const stored = localStorage.getItem("theme");
    70→      return (stored as Theme) || defaultTheme;
    71→    }
    72→    return defaultTheme;
    73→  });
    74→
    75→  useEffect(() => {
    76→    const root = document.documentElement;
    77→    if (theme === "dark") {
    78→      root.classList.add("dark");
    79→    } else {
    80→      root.classList.remove("dark");
    81→    }
    82→
    83→    if (switchable) {
    84→      localStorage.setItem("theme", theme);
    85→    }
    86→  }, [theme, switchable]);
    87→
    88→  const toggleTheme = switchable
    89→    ? () => {
    90→        setTheme(prev => (prev === "light" ? "dark" : "light"));
    91→      }
    92→    : undefined;
    93→
    94→  return (
    95→    <ThemeContext.Provider value={{ theme, toggleTheme, switchable }}>
    96→      {children}
    97→    </ThemeContext.Provider>
    98→  );
    99→}
   100→
   101→export function useTheme() {
   102→  const context = useContext(ThemeContext);
   103→  if (!context) {
   104→    throw new Error("useTheme must be used within ThemeProvider");
   105→  }
   106→  return context;
   107→}

   382→/** formats the local machine clock for the tray */
   383→function formatClock(date: Date): string {
   384→  return new Intl.DateTimeFormat(undefined, { hour: "2-digit", minute: "2-digit" }).format(date);
   385→}
   386→
   387→/** the tray clock, refreshed twice a minute */
   388→function useTrayClock(): string {
   389→  const [clock, setClock] = useState(() => formatClock(new Date()));
   390→  useEffect(() => {
   391→    const tick = window.setInterval(() => setClock(formatClock(new Date())), 30_000);
   392→    return () => window.clearInterval(tick);
   393→  }, []);
   394→  return clock;
   395→}
