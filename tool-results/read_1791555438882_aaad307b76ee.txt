   287→export function DevThinkApp({ os }: { os: OSHandle }) {
   288→  const meta = appMeta("devthink");
   289→  if (!meta) throw new Error("the devthink meta is missing from the catalog");
   290→  const [chatOpen, setChatOpen] = useState(false);
   291→  const page = meta.pages.some((p) => p.id === os.view.page) ? os.view.page : "projects";
   292→  const [projects, setProjects] = useStoredState<Project[]>("dt-projects-v1", [], isProjectList);
   293→
   294→  return (
   295→    <>
   296→      <AppHeader
   297→        app={meta}
   298→        active={page}
   299→        chatOpen={chatOpen}
   300→        onNavigate={(p) => os.openApp("devthink", p)}
   301→        onHome={os.goGateway}
   302→        onToggleChat={() => setChatOpen((v) => !v)}
   303→        theme={os.settings.theme}
   304→        onToggleTheme={os.toggleTheme}
   305→      />
   306→
   307→      <main className="shell">
   308→        {page === "aura" ? (
   309→          <div style={{ display: "flex", flexDirection: "column", flex: 1 }}>
   310→            <PageSection
   311→              eyebrow="chat · devthink.pro gateway"
   312→              title="Chat"
   313→              description="The gateway's intelligence, dedicated: ask about the platform, the engines or any app in the family."
   314→              reveal
   315→            />
   316→            <div
   317→              className="chat-panel"
   318→              style={{ width: "100%", maxWidth: 860, margin: "22px auto 0", height: "min(68vh, 620px)" }}
   319→            >
   320→              <AuraChat persona={PERSONAS.devthink} storageKey="dt-chat-devthink-v1" appLabel="devthink" />
   321→            </div>
   322→          </div>
   323→        ) : (
   324→          <div className={`app-layout${chatOpen ? " with-chat" : ""}`} style={{ marginTop: 26 }}>
   325→            <div>
   326→              {page === "projects" ? (
   327→                <ProjectsPage
   328→                  projects={projects}
   329→                  onCreate={(p) => setProjects((prev) => [p, ...prev])}
   330→                  onDelete={(id) => setProjects((prev) => prev.filter((p) => p.id !== id))}
   331→                />
   332→              ) : page === "history" ? (
   333→                <HistoryPage />
   334→              ) : page === "docs" ? (
   335→                <DocsPage />
   336→              ) : page === "explore" ? (
   337→                <ExplorePage os={os} />
   338→              ) : (
   339→                <SettingsPage os={os} />
   340→              )}
   341→            </div>
   342→            {chatOpen ? (
   343→              <div className="chat-panel">

   345→              </div>
   346→            ) : null}
   347→          </div>
   348→        )}
   349→      </main>
   350→    </>
   351→  );
   352→}

   354→/* ------------------------------ PROJECTS ------------------------------- */
   355→
   356→function ProjectsPage({
   357→  projects,
   358→  onCreate,
   359→  onDelete,
   360→}: {
   361→  projects: Project[];
   362→  onCreate: (p: Project) => void;
   363→  onDelete: (id: string) => void;
   364→}) {
   365→  const [query, setQuery] = useState("");
   366→  const [status, setStatus] = useState<"all" | ProjectStatus>("all");
   367→  const [open, setOpen] = useState(false);
   368→
   369→  const all = useMemo<Project[]>(() => [...projects, ...SEED_PROJECTS], [projects]);
   370→  const rows = useMemo(() => {
   371→    const q = query.trim().toLowerCase();
   372→    return all.filter(
   373→      (p) =>
   374→        (status === "all" || p.status === status) &&
   375→        (!q || p.name.toLowerCase().includes(q) || p.tag.toLowerCase().includes(q) || p.desc.toLowerCase().includes(q)),
   376→    );
   377→  }, [all, query, status]);
   378→
   379→  return (
   380→    <>
   381→      <PageSection
   382→        eyebrow="devthink.pro · platform"
   383→        title="Projects"
   384→        description="The provider-neutral workbench in one place: device-persisted local projects with status, tag and search. Creating here pushes an event into the OS History."
   385→        reveal
   386→      />
   387→
   388→      <div
   389→        className="row between"
   390→        style={{ marginTop: 26, marginBottom: 16 }}
   391→        role="toolbar"
   392→        aria-label="Project filters"
   393→      >
   394→        <div className="tabs" role="tablist" aria-label="Filter by status">
   395→          {(["all", "active", "paused", "done", "draft"] as const).map((s) => (
   396→            <button key={s} type="button" role="tab" aria-selected={status === s} onClick={() => setStatus(s)}>
   397→              {s === "all" ? "all" : STATUS_LABEL[s]}
   398→            </button>
   399→          ))}
   400→        </div>
   401→        <div className="row" style={{ gap: 10 }}>
   402→          <input
   403→            className="input"
   404→            type="search"
   405→            style={{ minWidth: 200, width: "auto" }}
   406→            placeholder="Search project, tag…"
   407→            aria-label="Search projects"
   408→            autoComplete="off"
   409→            value={query}
   410→            onChange={(e) => setQuery(e.target.value)}
   411→          />
   412→          <button type="button" className="btn" onClick={() => setOpen(true)}>
   413→            <Plus size={16} strokeWidth={1.8} aria-hidden="true" /> New project
   414→          </button>
   415→        </div>
   416→      </div>
   417→
   418→      <div className="grid cols-3" style={{ alignItems: "stretch" }}>
   419→        {rows.map((p, i) => (
   420→          <article key={p.id} className="glass glass-hover card proj reveal in">
   421→            <div className={`ph ${PH_CLASSES[i % PH_CLASSES.length]}`} aria-hidden="true" />
   422→            <div style={{ padding: 16 }}>
   423→              <div className="row between" style={{ marginBottom: 8 }}>
   424→                <h3 style={{ margin: 0, fontSize: "1.02rem" }}>{p.name}</h3>
   425→                <StatusDot label={STATUS_LABEL[p.status]} tone={STATUS_TONE[p.status]} pulse={p.status === "active"} />
   426→              </div>
   427→              <p style={{ margin: "0 0 12px", fontSize: ".9rem", color: "var(--sol-muted)" }}>{p.desc}</p>
   428→              <div className="row between">
   429→                <span className="badge">{p.tag}</span>
   430→                {projects.some((u) => u.id === p.id) ? (
   431→                  <button
   432→                    type="button"
   433→                    className="icon-btn"
   434→                    aria-label={`Remove project ${p.name}`}
   435→                    title="Remove project"
   436→                    onClick={() => {
   437→                      onDelete(p.id);
   438→                      pushOSEvent({ title: "Project removed — devthink", note: p.name, kind: "action" });
   439→                      toast("Project removed", { description: `${p.name} left this device.` });
   440→                    }}
   441→                  >
   442→                    <Trash2 size={16} strokeWidth={1.8} />
   443→                  </button>
   444→                ) : null}
   445→              </div>
   446→            </div>
   447→          </article>
   448→        ))}
   449→      </div>
   450→      {rows.length === 0 ? (
   451→        <div className="glass card" style={{ marginTop: 16 }}>
   452→          <p style={{ margin: 0, color: "var(--sol-muted)" }}>
   453→            No project matching “{query}”{status !== "all" ? ` in ${STATUS_LABEL[status]}` : ""} — adjust the filters.
   454→          </p>
   455→        </div>
   456→      ) : null}
   457→
   458→      <NewProjectModal
   459→        open={open}
   460→        onOpenChange={setOpen}
   461→        onCreate={(p) => {
   462→          onCreate(p);
   463→          pushOSEvent({ title: "Project created — devthink", note: `${p.name} · ${p.tag}`, kind: "action" });
   464→          toast.success(`Project “${p.name}” created`, {
   465→            description: `status ${STATUS_LABEL[p.status]} · tag ${p.tag}`,
   466→          });
   467→        }}
   468→      />
   469→    </>
   470→  );
   471→}

   473→function NewProjectModal({
   474→  open,
   475→  onOpenChange,
   476→  onCreate,
   477→}: {
   478→  open: boolean;
   479→  onOpenChange: (v: boolean) => void;
   480→  onCreate: (p: Project) => void;
   481→}) {
   482→  const [name, setName] = useState("");
   483→  const [desc, setDesc] = useState("");
   484→  const [tag, setTag] = useState("");
   485→  const [status, setStatus] = useState<ProjectStatus>("draft");
   486→
   487→  function submit() {
   488→    const n = name.trim();
   489→    if (!n) {
   490→      toast.error("Name required", { description: "Give the project a short name — everything else is optional." });
   491→      return;
   492→    }
   493→    onCreate({
   494→      id: `p-${Date.now()}`,
   495→      name: n,
   496→      desc: desc.trim() || "No description yet — edit it whenever you like.",
   497→      tag: tag.trim() || "general",
   498→      status,
   499→      at: Date.now(),
   500→    });
   501→    setName("");
   502→    setDesc("");
   503→    setTag("");
   504→    setStatus("draft");
   505→    onOpenChange(false);
   506→  }
   507→
   508→  return (
   509→    <Modal
   510→      open={open}
   511→      onOpenChange={onOpenChange}
   512→      title="New project"
   513→      description="The project persists on this device — nothing leaves the browser."
   514→    >
   515→      <div style={{ marginTop: 14 }}>
   516→        <div className="field">
   517→          <label htmlFor="np-name">Name</label>
   518→          <input
   519→            id="np-name"
   520→            className="input"
   521→            value={name}
   522→            maxLength={60}
   523→            autoComplete="off"
   524→            placeholder="e.g. Gateway normalizer"
   525→            onChange={(e) => setName(e.target.value)}
   526→          />
   527→        </div>
   528→        <div className="field">
   529→          <label htmlFor="np-desc">Description</label>
   530→          <textarea
   531→            id="np-desc"
   532→            className="input"
   533→            rows={3}
   534→            maxLength={200}
   535→            value={desc}
   536→            placeholder="One line is enough."
   537→            onChange={(e) => setDesc(e.target.value)}
   538→          />
   539→        </div>
   540→        <div className="row" style={{ alignItems: "flex-start" }}>
   541→          <div className="field" style={{ flex: "1 1 140px" }}>
   542→            <label htmlFor="np-tag">Tag</label>
   543→            <input
   544→              id="np-tag"
   545→              className="input"
   546→              value={tag}
   547→              maxLength={20}
   548→              placeholder="gateway · cli · web"
   549→              onChange={(e) => setTag(e.target.value)}
   550→            />
   551→          </div>
   552→          <div className="field" style={{ flex: "1 1 160px" }}>
   553→            <label htmlFor="np-status">Status</label>
   554→            <select
   555→              id="np-status"
   556→              className="input"
   557→              value={status}
   558→              onChange={(e) => setStatus(e.target.value as ProjectStatus)}
   559→            >
   560→              {(["draft", "active", "paused", "done"] as const).map((s) => (
   561→                <option key={s} value={s}>
   562→                  {STATUS_LABEL[s]}
   563→                </option>
   564→              ))}
   565→            </select>
   566→          </div>
